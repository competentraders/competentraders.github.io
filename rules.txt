
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 1. USERS: Updated to allow public email checks for recovery
    match /users/{userId} {
      // Allows the Recovery Scan to find the email without being logged in
      allow read: if true; 
      // Only the logged-in owner can change their balance or settings
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // 2. DEPOSITS
    match /deposits/{depositId} {
      allow create: if request.auth != null;
      allow read: if request.auth != null;
    }

    // 3. WITHDRAWALS
    match /withdrawals/{withdrawId} {
      allow create: if request.auth != null;
      allow read: if request.auth != null;
    }

    // 4. FOOTBALL PREDICTIONS
    match /predictions/{document=**} {
      allow read, write: if true; 
    }

    // 5. FOREX PLANS
    match /forex_plans/{document=**} {
      allow read, write: if true; 
    }

    // 6. ROI PLANS
    match /roi_plans/{document=**} {
      allow read, write: if true;
    }

    // 7. ACTIVE INVESTMENTS
    match /active_investments/{document=**} {
      allow read, write: if request.auth != null;
    }

    // 8. VIP PLANS
    match /vip_plans/{document=**} {
      allow read: if request.auth != null;
      allow write: if true; 
    }
  }
}
